# 入站允许规则。只要出现 [allow_inbound]，未匹配的来源 IP 一律拒绝。
# 支持精确 IPv4/IPv6 和 CIDR；空配置段表示拒绝全部入站连接。
[allow_inbound]
# 203.0.113.10
# 203.0.113.0/24
# 2001:db8:100::/48

# 入站拒绝规则。deny 优先；! 可为较宽的拒绝网段添加例外。
[deny_inbound]
# 203.0.113.66
# 10.0.0.0/8
# !10.1.0.0/16

# 出站允许规则。只要出现 [allow_outbound]，未匹配的 host+port 一律拒绝。
# 支持精确 IP/域名、单端口和闭区间端口范围；IPv6 必须使用方括号。
[allow_outbound]
# api.example.com:443
# 192.168.1.30:8000-8999
# [2001:db8::20]:10000-10100

# 出站拒绝规则（包含域名、IP 和 CIDR）。deny 优先。
[deny_outbound]
127.0.0.0/8
10.0.0.0/8
192.168.0.0/16
172.16.0.0/12
::1
::/0
deny-example.com
*.deny-example.com
